```python
def first_last(s): return s[0] + s[-1]
```

```python
def avg4(a,b,c,d): return (a+b+c+d)/4
```

```python
def strong(n): return "да" if n%3==0 and n%5==0 else "нет"
```

```python
def count_upper(t): return sum(ch.isupper() for ch in t)
```

```python
def product_digits(n):
    if n<10: return n
    return (n%10) * product_digits(n//10)
```

```python
def remove_vowels(s):
    if not s: return ""
    return ("" if s[0] in "aeiouаеёиоуыэюя" else s[0]) + remove_vowels(s[1:])
```

```python
def list_reverse(lst):
    if not lst: return []
    return list_reverse(lst[1:]) + [lst[0]]
```

```python
def uniq_count(lst):
    if not lst: return 0
    return (0 if lst[0] in lst[1:] else 1) + uniq_count(lst[1:])
```

```python
def f(words): return list(map(lambda w: len(w), words))
```

```python
def f(nums): return list(filter(lambda x: x%10==7, nums))
```

```python
def flatten_once(m): return [x for row in m for x in row]
```

```python
def sum_even_digits(n):
    if n<10: return n if n%2==0 else 0
    d=n%10
    return (d if d%2==0 else 0)+sum_even_digits(n//10)
```

```python
def clean_spaces(s):
    if len(s)<2: return s
    if s[0]==" " and s[1]==" ":
        return clean_spaces(s[1:])
    return s[0] + clean_spaces(s[1:])
```

```python
def is_sorted_rec(lst):
    if len(lst)<=1: return True
    return lst[0]<=lst[1] and is_sorted_rec(lst[1:])
```

```python
import math
def perm_count(s):
    from collections import Counter
    c=Counter(s)
    n=len(s)
    res=math.factorial(n)
    for v in c.values():
        res//=math.factorial(v)
    return res
```

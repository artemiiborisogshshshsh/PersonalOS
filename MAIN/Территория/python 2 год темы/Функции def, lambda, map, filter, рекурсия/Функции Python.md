# 1. Чётное или нечётное
def even_or_odd(n):
    if n % 2 == 0:
        return "чётное"
    else:
        return "нечётное"
print(even_or_odd(12))  # 👉 чётное

# 2. Среднее трёх чисел
def average_of_three(a, b, c):
    return (a + b + c) / 3
print(average_of_three(3, 6, 9))  # 👉 6.0

# 3. Длина слова
def word_length(word):
    return len(word)
print(word_length("питон"))  # 👉 5

# 4. Минимум и максимум
def min_and_max(numbers):
    return min(numbers), max(numbers)
print(min_and_max([3, 8, 1, 10, 5]))  # 👉 (1, 10)

# 5. Квадрат списка
def squares(numbers):
    result = []
    for n in numbers:
        result.append(n ** 2)
    return result
print(squares([1, 2, 3, 4]))  # 👉 [1, 4, 9, 16]

# 6. Сумма чётных чисел
def sum_of_evens(numbers):
    total = 0
    for n in numbers:
        if n % 2 == 0:
            total += n
    return total
print(sum_of_evens([1, 2, 3, 4, 5, 6]))  # 👉 12

# 7. Замена пробелов
def replace_spaces(s):
    return s.replace(" ", "_")
print(replace_spaces("я учу питон"))  # 👉 я_учу_питон

# 8. Первая и последняя буква
def first_and_last(word):
    if len(word) == 0:
        return ""
    return word[0] + word[-1]
print(first_and_last("python"))  # 👉 pn

# 9. Счётчик слов
def word_count(sentence):
    return len(sentence.split())
print(word_count("я люблю питон"))  # 👉 3

# 10. Список от 1 до N
def list_to_n(n):
    return list(range(1, n + 1))
print(list_to_n(5))  # 👉 [1, 2, 3, 4, 5]

# 1. Перевёртыш слов
def reverse_words(sentence):
    return " ".join(sentence.split()[::-1])
print(reverse_words("Я люблю питон"))  # 👉 питон люблю Я

# 2. Цифровой след
def digital_root(n):
    while n >= 10:
        s = 0
        for digit in str(n):
            s += int(digit)
        n = s
    return n
print(digital_root(9875))  # 👉 2

# 3. Шифр «перестановка»
def shuffle_string(s):
    even = ""
    odd = ""
    for i in range(len(s)):
        if i % 2 == 0:
            even += s[i]
        else:
            odd += s[i]
    return even + " " + odd
print(shuffle_string("python"))  # 👉 pto yhn

# 4. Лестница из символов
def draw_stairs(symbol, height):
    for i in range(1, height + 1):
        print(symbol * i)
draw_stairs("#", 4)

# 5. Функция «Эхо»
def echo(word, n):
    result = []
    for i in range(1, n + 1):
        result.append(word + "!" * i)
    return " ".join(result)
print(echo("Привет", 3))  # 👉 Привет! Привет!! Привет!!!

# 6. Счётчик букв
def count_letter(s, letter):
    return s.count(letter)
print(count_letter("banana", "a"))  # 👉 3

# 7. Таблица умножения
def multiplication_table(n):
    for i in range(1, 11):
        print(f"{n} x {i} = {n * i}")
multiplication_table(5)

# 8. Магический переворот
def reverse_number(n):
    return int(str(n)[::-1])
print(reverse_number(1234))  # 👉 4321

# 9. Игра «Фантазёр»
import random
def fun_name(name):
    adjectives = ["Молния", "Смешинка", "Тигр", "Улыбка", "Ракета"]
    return name + "-" + random.choice(adjectives)
print(fun_name("Катя"))

# 10. Функция-считалочка
def counting(n):
    return ", ".join(str(i) for i in range(1, n + 1))
print(counting(5))  # 👉 1, 2, 3, 4, 5
